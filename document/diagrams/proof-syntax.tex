\begin{grammar}\footnotesize
	<proof> ::=  <version><proofCommand>$^*$ 
	
	<proofCommand> ::= <assumeNetwork>
	\alt <assume>
	\alt<step>
	
	%<name> ::= [A-Za-z][\_A-Za-z0-9]*
	<assumeNetwork> ::=  (\texttt{assume-network} <name> <assume>$^*$)
	
	
	<assume> ::= (\texttt{assume} <name> <bool> (\texttt{:layer} <name>)$^?$)

	<subproof> ::= (\texttt{anchor} :step <name>)
	
	<step> ::= (\texttt{step} <name> <clause> \texttt{:rule} <rule\_name> <premisesAnnotation>$^?$ <argsAnnotation>$^?$ )
	
	<clause> ::= (\texttt{cl} <bool>$^*$)
	
	<premisesAnnotation> ::= \texttt{:premises} (<name>$^+$)
	
	<argsAnnotation> ::= \texttt{:args} (<rational>$^+$)
	
	<rational> ::= -$^?$[0-9]$^+$/$^?$[0-9]$^+$ 
	\alt -$^?$[0-9]$^+$(.[0-9]$^+$)$^?$
		
	<bool> ::= (\texttt{and} <bool> <bool>$^+$)
	\alt (\texttt{or} <bool> <bool>$^+$)
	\alt (\texttt{not} <bool>)
	\alt (\texttt{ite} <bool> <bool> <bool>)
	\alt (\texttt{\textless} <arith> <arith>)
	\alt (\texttt{\textless=} <arith> <arith>)
	\alt (\texttt{\textgreater} <arith> <arith>)
	\alt (\texttt{\textgreater=} <arith> <arith>)
	\alt (\texttt{==} <arith> <arith>)
	\alt (\texttt{!=} <arith> <arith>)
	\alt (\texttt{is\_int} <arith>)
	
	<rule\_name> ::=  \texttt{false}
	\alt  \texttt{not\_not}
	\alt \texttt{resolution}
	\alt \texttt{or}
	\alt \texttt{and\_pos} 
	\alt \texttt{and\_neg}
	\alt \texttt{ite1} 
	\alt \texttt{ite2}
	\alt \texttt{la\_totality}
	\alt \texttt{la\_tautology}
	\alt \texttt{la\_generic}
	\alt \texttt{hole}
	%<arith> ::= <element>$\networkTheoryVarParam$
	%\alt <name> <indices>$^?$
	%\alt (\texttt{-} <arith>)
	%\alt (\texttt{+} <arith> <arith>$^+$)
	%\alt (\texttt{*} <arith> <arith>$^+$)
	%\alt (\texttt{-} <arith> <arith>$^+$)
	
	%<indices> ::= [i, ..., i] for $i \in \mathbb{N}$
\end{grammar}
